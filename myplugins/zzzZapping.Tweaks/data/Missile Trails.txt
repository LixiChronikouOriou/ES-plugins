# Adds smoke/flame/energy trails to most homing weapons and some larger ballistic or artillery weapons.
# This edited version
# -- Addresses only vanilla outfits,
# -- Updates the spark counts after the mass changes of merged PR 12239,
# -- Streamlines some trails.


effect "gatling smoke"
	sprite "effect/smoke"
		"scale" .05
		"no repeat"
		"frame rate" 20
	"lifetime" 2
	"random spin" 6
	"random angle" 360
	"random velocity" 1
	"velocity scale" .5
